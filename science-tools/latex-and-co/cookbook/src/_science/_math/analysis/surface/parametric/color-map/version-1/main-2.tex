\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw*}{name = param-surf-colmap-Y}
local ld = luadraw

local M = ld.pt3d.M

local palext = require 'luadraw_palettes'

local pal = palext.BurningGrass

local pi  = math.pi
local sin = math.sin
local cos = math.cos

local facets = ld.surface(
  function(u, v)
    return M(
      3*v*cos(2*u),
      3*v*sin(2*u),
      3*v*cos(u)
    )
  end,
  0, 2*pi, 0, 1,
  {50, 30}
)

local graphview = ld.graph3d:new{
  viewdir = {170, 80},
  bbox    = false
}

------
-- Un seul ¨chgt ici : usage de la méthode ''"y"''.
------
graphview:Dpoly(
  ld.facet2poly(facets),
  {
    usepalette = {pal, "y"},
    mode       = ld.mShaded
  }
)

graphview:Show()
\end{luadraw*}

\end{document}
