\usepackage{amsmath}

%%%

\newcommand{\test}%
           {A \eqdot B}

\newcommand{\verbtest}%
           {A \mathmacro{eqdot} B}

\NewDocumentCommand{\mathmacro}{m}%
                   {\;\backslash#1\;}

\NewDocumentCommand{\txtmacro}{m}%
                   {\mathtt{\backslash#1}}

\newcommand{\testmathstyle}{
  \[
    \renewcommand{\arraystretch}{1.2}%
    \begin{array}{lll}
        \textbf{Style}
      & \mathtt{\verbtest}
      \\ \hline
        \txtmacro{displaystyle}
      & \displaystyle \test
      \\
        \txtmacro{textstyle}
      & \textstyle \test
      \\
        \txtmacro{scriptstyle}
      & \scriptstyle \test
      \\
        \txtmacro{scriptscriptstyle}
      & \scriptscriptstyle \test
    \end{array}
  \]
}
