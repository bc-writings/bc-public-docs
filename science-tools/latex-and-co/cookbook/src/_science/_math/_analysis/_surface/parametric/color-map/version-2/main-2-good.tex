\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw*}{name = param-surf-colmap-OK}
local ld = luadraw

local M = ld.pt3d.M

local palext = require 'luadraw_palettes'

local pal = palext.Gnuplot

local pi   = math.pi
local sin  = math.sin
local cos  = math.cos
local tanh = math.tanh

------
-- Points et vecteurs utilisés pour compléter le tracé via
-- des symétries.
------
local Origin = ld.pt3d.Origin
local vecJ   = ld.pt3d.vecJ
local vecK   = ld.pt3d.vecK


------
-- Nous gardons juste un quart de la ¨surf qui n'a que
-- des faces orientées vers l'observateur.
------
local facets = ld.surface(
  function(u, v)
    return M(
      2*(1 + cos(v))*cos(u),
      2*(1 + cos(v))*sin(u),
      2*sin(v)*tanh(u - pi)
    )
  end,
  pi, 0, 0, pi,
  {25, 15}
)

------
-- Le reste de la surface s'obtient via des symétries, mais
-- en prenant garde à donner la bonne orientation aux faces
-- visibles.
------
facets = ld.concat(
  facets,
  ld.reverse_face_orientation(
    ld.sym3d(
      facets,
      {Origin, vecJ}
    )
  )
)

facets = ld.concat(
  facets,
  ld.reverse_face_orientation(
    ld.sym3d(
      facets,
      {Origin, vecK}
    )
  )
)

------
-- Rien de neuf pour la fin...
------
local graphview = ld.graph3d:new{
  viewdir = {120, 55},
  bbox    = false
}

graphview:Dpoly(
  ld.facet2poly(facets),
  {
    usepalette = {pal, "z"},
    mode       = ld.mShaded
  }
)

graphview:Show()
\end{luadraw*}

\end{document}
