% !TEX TS-program = lualatex

\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw*}{name = orientation-n-box-axes-partial}
local ld = luadraw

local M = ld.pt3d.M

local pi   = math.pi
local sin  = math.sin
local cos  = math.cos
local tanh = math.tanh

local graphview = ld.graph3d:new{
  window3d = {-3, 6, -4, 6, -2, 2},
  viewdir  = {190, 80},  -- Un autre point de vue pour mieux
                         -- visualiser le ¨chgt d'orientation.
  adjust2d = true,
  size     = {14, 14},
  bbox     = false
}

graphview:Dboxaxes3d({
  grid      = true,
  gridcolor = "gray",
  fillcolor = "lightgray"
})

------
-- Nous ne gardons qu'une partie de la surface.
------
local facets = ld.surface(
  function(u, v)
    return M(
      2*(1 + cos(v))*cos(u),
      2*(1 + cos(v))*sin(u),
      2*sin(v)*tanh(u-pi)
    )
  end,
  0, 2*pi, 0.3, pi,
  {51, 15}
)

graphview:Dfacet(
  facets,
  {
    mode  = ld.mShadedOnly,
    color = "ForestGreen"
  }
)

graphview:Show()
\end{luadraw*}

\end{document}
