\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}
\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw*}{name = gamma-modulus-surface-optimized}
local palext = require 'luadraw_palettes'

local pal = palext.ShiftRainbow

local ld = luadraw

local M   = ld.pt3d.M
local cpx = ld.cpx
local Z   = cpx.Z

local vecK = ld.pt3d.vecK

local graphview = ld.graph3d:new{
  window3d = {-4, 5, -5, 5, 0, 6},
  viewdir  = {-120, 60},
  adjust2d = true,
  bbox     = false
}

local gamma = function(x, y)
  local k_facto, sign, z = 1, 1, Z(x, y)

  local S = 1 / z

  for k = 1, 50 do
    sign    = -sign
    k_facto = k_facto*k

    S = S + sign / k_facto / (z + k)
  end

  return S + ld.int(
    function(t)
      return cpx.exp((z - 1)*math.log(t) - t)
    end,
    1, 10
  )
end

------
-- Construction de la ¨surf "complexe" à la main en mémorisant
-- les points utilisés.
------
local memo_cpx_args = {}
local key

local xykey = function(x, y)
    return x .. "/" .. y
end

local gamma_modulus_pt3d = function(x, y)
  local z = gamma(x, y)

  memo_cpx_args[xykey(x, y)] = cpx.arg(z)

  return M(x, y, cpx.abs(z))
end

------
-- Coloration en ¨fonc de l'¨arg principal du complexe
-- `gamma(z)` via le minimum des arguments associés aux
-- sommets de la face.
------
local color_coef = function(facet)
  local miniarg = math.huge

  for _ , A in ipairs(facet) do
    key = xykey(A.x, A.y)

    local arg = memo_cpx_args[key]

    if arg ~= nil then
      if miniarg > arg then
        miniarg = arg
      end
    end
  end

  return (miniarg + math.pi) / (2*math.pi)
end

------
-- Rien de neuf pour la fin...
------
local cpx_surf = ld.surface(
  gamma_modulus_pt3d,
  -4, 5, -5, 5,
  {50, 50}
)

cpx_surf = ld.cutfacet(
  cpx_surf,
  {6*vecK, -vecK}
)

cpx_surf = graphview:Sortfacet(cpx_surf)

graphview:Dboxaxes3d({
  grid      = true,
  gridcolor = "gray",
  fillcolor = "LightGray"
})

for _ , facet in ipairs(cpx_surf) do
  graphview:Dpolyline3d(
    facet,
    true,
       "fill = "
    .. ld.palette(
          pal,
          color_coef(facet)
       )
  )
end

graphview:Show()
\end{luadraw*}

\end{document}
