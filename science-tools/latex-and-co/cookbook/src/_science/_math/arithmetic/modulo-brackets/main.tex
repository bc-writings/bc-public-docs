\documentclass[varwidth, border = 3pt]{standalone}

\usepackage{main}

%%%
% Nous utilisons les macros ''\pod'' et ''\substack'' de
% ''amsmaths''.
%%%
\usepackage{amsmath}


%%%
% prototype::
%    #1 : une \1ere \cmd de \mef d'un unique matéruiel \math.
%    #2 : une \2nde \cmd de \mef d'un unique matéruiel \math.
%
%    :action: cette macro simplifie la rédaction des tests.
%%%
\NewDocumentCommand{\test}{ m m }{
  $a \equiv b + c #1{n}$
  \, et \,
  $a \equiv b + c #2{n}$

  \smallskip

  $\displaystyle\sum_{\substack{p \le x \\ p \equiv 1 #1{4}}} f(p)$
  \, et
  $\displaystyle\sum_{\substack{p \le x \\ p \equiv 1 #2{4}}} f(p)$
}

\begin{document}

\textbf{amsmath.}
Usage de \verb+\pmod+ et \verb +\pod+.

\test{\pmod}{\pod}


\bigskip


\textbf{Solutions "maison".}
Usage de \verb+\brmod+ et \verb +\brod+.

\test{\brmod}{\brod}

\end{document}
