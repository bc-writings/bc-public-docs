\documentclass[varwidth, border = 3pt]{standalone}

%%%
% Nous déléguons tout le travail au package ''ibrackets'' qui redéfinit
% le comportement des crochets en mode \math.
%%%
\usepackage{ibrackets}

%%%
% Nous allons utiliser la macro ''\dfrac'' proposée par ''amsmath''.
%%%
\usepackage{amsmath}

\begin{document}

$[a, b]$
\quad
$]a, b[$
\quad
$]\dfrac{1}{2}, \dfrac{2}{3}[$

$]-\infty, b]$
\quad
$]a, +\infty[$
\quad
$[a, \infty[$

$]-\infty, \infty[$
\quad
$]-\infty, +\infty[$

$]1, 2[ \cup [2, 3[ = ]1, 3[$


\medskip


Avec un autre séparateur :

$[a ; b]$
\quad
$]a .. b]$
\quad
$]-\infty .. +\infty[$


\medskip


Un test hors intervalle : 

$a = b \, [p]$

\end{document}
