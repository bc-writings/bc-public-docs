\documentclass[varwidth, border = 10pt]{standalone}

\usepackage{main}

%%%
% Nous allons utiliser la macro ''\dfrac'' proposée par ''amsmath''.
%%%
\usepackage{amsmath}

%%%
% Nous présentons deux réglages dans deux colonnes côte-à-côte.
%%%
\usepackage{multicol}

%%%
% prototype::
%     #1 : \delim ouvrant à droite pour tester l'écriture directe.
%     #2 : \delim ouvrant à gauche pour tester l'écriture directe.
%%%
\NewDocumentCommand{\example}{ m m }{
    Usages classiques :

    $\interval{a, b}$
    \quad
    $\interval[big, oc]{a, b}$

    $\interval[co]{\dfrac{1}{2}, \dfrac{2}{3}}$
    \quad
    $\interval[Big, c]{\dfrac{1}{2}, \dfrac{2}{3}}$
    \quad
    $\interval[auto]{\dfrac{1}{2}, \dfrac{2}{3}}$

    $\interval[oc]{-oo, b}$
    \quad
    $\interval{a, +oo}$
    \quad
    $\interval[co]{a, oo}$

    $\interval{-oo, oo}$
    \quad
    $\interval{-oo, +oo}$

    $\interval{1, 2} \cup \interval[co]{2, 3} = \interval{1, 3}$

    \medskip

    Avec un autre séparateur :

    $\interval[c, sep=;]{a, b}$
    \quad
    $\interval[oc, sep=..]{a, b}$
    \quad
    $\interval[sep=..]{-oo, +oo}$

    \medskip

    À la main :

    $#11, 2#2 \cup [2, 3#2 = #11, 3#2$
}

\begin{document}

\begin{multicols}{2}
  \paragraph{À l'américaine}
  \leavevmode

  \example{(}{)}

  \paragraph{À la française}
  \leavevmode

  \intervalsetup{french}
  \example{]}{[}
\end{multicols}

\end{document}
