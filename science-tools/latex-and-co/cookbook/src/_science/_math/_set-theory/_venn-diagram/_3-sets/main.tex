\documentclass[border = 3pt]{standalone}

%%%
% Des couleurs nommées faciles d'emploi via le package \xcolor!
%%%
\usepackage[svgnames]{xcolor}

%%%
% La bibliothèque \luadraw allie une facilité d’utilisation
% à un rendu particulièrement soigné.
%%%
\usepackage{luadraw}

\begin{document}

\begin{luadraw*}{name = 3-sets}
local ld = luadraw

local cpx = ld.cpx

------
-- ¨Def de la zone graphique.
------
local graphview = ld.graph:new{
  window = {0, 16, 0, 4.5},
  bbox   = false
}

------
-- Taille de la police des étiquettes.
------
graphview:Labelsize("footnotesize")

------
-- Nos ingrédients de base.
------
local i = cpx.I

local U = {
  -6 - 6*i, 6 - 6*i, 6 + 6*i, -6 + 6*i,
  "l",
  "cl"
}

local A = ld.set(i, 0)
local B = ld.set(-2 - i, 25)
local C = ld.set(2 - i, -25)

-- Conversion des chemins en listes de complexes.
local C1, C2, C3 = ld.path(A), ld.path(B), ld.path(C)

------
-- Deux fonctions de tracé pour factoriser le code.
------
local drawdiag = function(paths, legend)
  graphview:Dpolyline(
    paths,
    true,
       "draw = none,"
    .. "fill = lightgray,"
    .. "even odd rule"
  )

  graphview:Dpath(U)
  graphview:Dpath(A)
  graphview:Dpath(B)
  graphview:Dpath(C)

  graphview:Dlabel(
    "$U$", -6.5 + 6*i,
    {pos = "NE"},
    "$A$", 4.75*i,
    {pos = "N"},
    "$B$", -4 + 3*i,
    {pos = "W"},
    "$C$", 4 + 3*i,
    {pos = "E"},
    legend, -6.5 - 6.75*i,
    {}
  )
end

local addcase = function(
    xvpmin, xvmax, yvpmin, yvpmax,
    paths,
    legend
)
  graphview:Saveattr()
    graphview:Viewport(xvpmin, xvmax, yvpmin, yvpmax)
    graphview:Coordsystem(-7, 7, -7, 7)

    drawdiag(paths, legend)
  graphview:Restoreattr()
end

------
-- Tracé de 4 cas.
------
addcase(
  0, 4, 0, 4,
  {
    ld.setminus(ld.cap(C1, C2), C3),
    ld.setminus(ld.cap(C1, C3), C2),
    ld.setminus(ld.cap(C3, C2), C1)
  },
  "a...."
)

addcase(
  4, 8, 0, 4,
  {
    ld.setminus(C1, C2),
    ld.setminus(C2, C1)
  },
  "b...."
)

addcase(
  8, 12, 0, 4,
  {
    ld.setminus(C1, ld.cup(C2, C3)),
    ld.setminus(C3, ld.cup(C1, C2))
  },
  "c...."
)

addcase(
  12, 16, 0, 4,
  ld.concat(
    ld.path(U)[1],
    ld.cup(C1, ld.cup(C2, C3))
  ),
  "d...."
)

------
-- Montrons le résultat de notre oeuvre.
------
graphview:Show()
\end{luadraw*}

\end{document}
