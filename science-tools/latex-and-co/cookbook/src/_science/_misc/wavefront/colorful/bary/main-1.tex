% !TEX TS-program = lualatex

\documentclass[border = 3pt]{standalone}

\usepackage[svgnames]{xcolor}

\usepackage[3d]{luadraw}

\begin{document}

\begin{luadraw*}{name = z-bary-colmap}
local palext = require 'luadraw_palettes'

local ld = luadraw

------
-- Récupération du modèle 3D au format Wavefront.
------
local vdir = {35, 60}
local file = "../../models/nefertiti.obj"

local polyhedron, bounding_box = ld.read_obj_file(file)

------
-- ¨Def de la zone graphique.
------
local graphview = ld.graph3d:new{
  window3d = bounding_box,
  viewdir  = vdir,
  adjust2d = true,
  bbox     = false
}

------
-- Coloration liée à la côte du barycentre de la facette.
------
local pal = palext.BlindFish

graphview:Dpoly(
  polyhedron,
  {
    usepalette = {pal, "z"},
    mode       = ld.mShadedOnly
  }
)

------
-- Montrons le résultat de notre oeuvre.
------
graphview:Show()
\end{luadraw*}

\end{document}
