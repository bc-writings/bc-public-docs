\documentclass[varwidth, border = 3pt]{standalone}

\usepackage{main}

%%%
% Nous allons utiliser les macros ''\mathbb'' et ''\mathfrak'' proposées
% par ''amsfonts''.
%%%
\usepackage{amsfonts}

%%%
% Définition de ''\NN'', ''\QQ'' et ''\ZZ''.
%%%
\makeabbrev{\mathbb}{#1#1}{N, Q, Z}

%%%
% Définition de ''\frC'' et ''\frgl''.
%%%
\makeabbrev{\mathfrak}{fr#1}{C, gl}

\begin{document}

\verb#$\NN$#, \verb#$\QQ$# et \verb#$\ZZ$# donnent
$\NN$, $\QQ$ et $\ZZ$.

\verb#$\frC$# et \verb#$\frgl$# donnent 
$\frC$ et $\frgl$.

\end{document}
