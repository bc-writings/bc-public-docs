\documentclass[varwidth, border = 3pt]{standalone}

\usepackage{main}

%%%
% ''amsfonts'' fournit les macros ''\mathbb'' et ''\mathfrak''.
%%%
\usepackage{amsfonts}

%%%
% Définition de ''\NN'', ''\QQ'' et ''\ZZ'' comme abréviations
% de ''\mathbb{N}'', ''\mathbb{Q}'' et ''\mathbb{Z}''.
%%%
\makeabbrev{\mathbb}{#1#1}{N, Q, Z}

%%%
% Définition de ''\frC'' et ''\frgl'' comme abréviations de
% ''\mathfrak{C}'' et ''\mathfrak{gl}''.
%%%
\makeabbrev{\mathfrak}{fr#1}{C, gl}

\begin{document}

\verb#$\NN$#, \verb#$\QQ$# et \verb#$\ZZ$# donnent
$\NN$, $\QQ$ et $\ZZ$.

\verb#$\frC$# et \verb#$\frgl$# donnent 
$\frC$ et $\frgl$.

\end{document}
