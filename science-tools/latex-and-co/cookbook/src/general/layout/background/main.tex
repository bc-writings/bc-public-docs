\documentclass[a4paper]{article}

%%%
% Un \pack pour obtenir du faux contenu.
%%%
\usepackage{lipsum}

%%%
% La \bib ''luadraw'' allie une facilité d’utilisation à un
% rendu particulièrement soigné.
%%%
\usepackage{luadraw}

%%%
% \Def d'une \fonc \lua pour dessiner, via \luadraw, la même
% suite de cercles au fond de chaque page.
%%%
\begin{luacode*}
function background()
  local ld = luadraw

------
-- ¨Def de la zone graphique.
------
  local graphview = ld.graph:new{
    margins = {0, 0, 0, 0},
    size    = {18, 18}
  }

------
-- Réglages liés aux lignes.
------
  graphview:Linecolor("lightgray")

------
-- Traçons des cercles.
------
  _Zp = ld.cpx.Zp

  for j = 1, 5 do
    for k = 1, 10*j do
      graphview:Dcircle(
        _Zp(j, k * math.pi / (5*j)),
        1
      )
    end
  end

------
-- On renvoie directement le code ¨tikz pour l'utiliser où l'on
-- souhaite.
------
  graphview:Sendtotex()
end
\end{luacode*}

%%%
% Création d'une macro sans \arg qui sera utilisée lors de la \prod
% du fond de chaque page.
%%%
\newcommand\background{\directlua{background()}}

%%%
% Utilisation d'un point d'ancrage du noyau \latex pour agir lors de
% la fabrication des fonds des pages.
%%%
\AddToHook{shipout/background}{%
% Un dessin \tikz.
    \put(\paperwidth/2 - 9cm, -\paperheight/2 - 9cm){%
        \background%
    }
% Si besoin, le \num de page est utilisable.
    \put(\paperwidth/2 - 1.5em, -\paperheight/2){%
        \Huge\bfseries\color{red}\fbox{\thepage}%
    }%
}
